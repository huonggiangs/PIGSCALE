cmake_minimum_required(VERSION 3.16)
project(pig_host_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -finput-charset=UTF-8 -fexec-charset=UTF-8")
set(IDF_PATH "C:/Espressif/frameworks/esp-idf-v5.5.5" CACHE PATH "ESP-IDF source")
set(LVGL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../managed_components/lvgl__lvgl")
set(APP "${CMAKE_CURRENT_SOURCE_DIR}/../../main")
add_executable(protocol_test protocol_test.c ${APP}/network/camera_protocol.c ${IDF_PATH}/components/json/cJSON/cJSON.c)
target_include_directories(protocol_test PRIVATE ${APP}/network ${IDF_PATH}/components/json/cJSON)
add_test(NAME camera_protocol COMMAND protocol_test)

file(GLOB_RECURSE LVGL_SRC "${LVGL_DIR}/src/*.c")
add_library(lvgl STATIC ${LVGL_SRC})
target_include_directories(lvgl PUBLIC ${LVGL_DIR})
target_compile_definitions(lvgl PUBLIC LV_CONF_SKIP LV_USE_OS=0 LV_USE_LOG=0 LV_COLOR_DEPTH=32
    LV_FONT_MONTSERRAT_40=1 LV_FONT_MONTSERRAT_48=1 LV_USE_TJPGD=1 LV_USE_FS_MEMFS=1 LV_FS_MEMFS_LETTER=77)
file(GLOB FONTS "${APP}/ui/fonts/*.c")
file(GLOB ICONS "${APP}/ui/icons/*.c")
add_executable(settings_test settings_test.c ${APP}/ui/ui_common.c ${FONTS} ${ICONS})
target_include_directories(settings_test PRIVATE stubs ${APP}/ui ${APP}/network)
target_compile_definitions(settings_test PRIVATE LV_LVGL_H_INCLUDE_SIMPLE)
target_link_libraries(settings_test lvgl)
add_test(NAME settings_keyboard COMMAND settings_test)
set_tests_properties(settings_keyboard PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
